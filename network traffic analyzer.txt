
from scapy.all import sniff, IP, TCP, UDP, ICMP

def analyze_packet(packet):
    if IP in packet:
        source = packet[IP].src
        destination = packet[IP].dst
        size = len(packet)

        if TCP in packet:
            protocol = "TCP"
        elif UDP in packet:
            protocol = "UDP"
        elif ICMP in packet:
            protocol = "ICMP"
        else:
            protocol = "Other"

        print("\n--- Network Traffic Details ---")
        print("Source IP      :", source)
        print("Destination IP :", destination)
        print("Protocol       :", protocol)
        print("Packet Size    :", size, "bytes")

print("Network Traffic Analyzer Started...")
print("Capturing network packets...")

# Capture 10 packets
sniff(prn=analyze_packet, count=10, store=False)
